window.recordAction = async function(actionName, scoreChange = 0, attendanceType = null) { if(!currentStudentId) return; closeActionModal(); const student = allStudents.find(s => s.id === currentStudentId); if(student) { const now = new Date(); const dayName = now.toLocaleDateString('ar-SA', { weekday: 'long' }); const dateFormatted = now.toLocaleDateString('ar-SA', { year: 'numeric', month: '2-digit', day: '2-digit' }); const timeFormatted = now.toLocaleTimeString('ar-SA', { hour: '2-digit', minute: '2-digit' }); const formattedTime = `${dayName} | ${dateFormatted} - ${timeFormatted}`; const todayDateOnly = dateFormatted; // تاريخ اليوم فقط للمقارنة let attendanceStats = student.attendanceStats || { absents: 0, lates: 0, excuses: 0 }; // التحقق من الحضور وتحديث العدّادات وحذف التعارض في نفس اليوم let actionsList = student.actions || []; if(attendanceType) { // إذا تم رصد حضور أو تأخير أو استذان، نقوم بإزالة أي حالة غياب سابقة مسجلة في نفس اليوم if(attendanceType === 'حاضر' || attendanceType === 'متأخر' || attendanceType === 'مستأذن' || attendanceType === 'عن بعد') { actionsList = actionsList.filter(act => { let isSameDay = act.timestamp.includes(todayDateOnly); let isAbsence = act.title.includes('غائب'); if(isSameDay && isAbsence) { // تخفيض عداد الغياب إذا تم حذفه if(attendanceStats.absents > 0) attendanceStats.absents--; return false; // حذف سجل الغياب لنفس اليوم } return true; }); } // تحديث عدادات الحضور والغياب if(attendanceType.includes('غائب') || attendanceType === 'غائب' || attendanceType === 'غائب بعذر') attendanceStats.absents++; if(attendanceType === 'متأخر') attendanceStats.lates++; if(attendanceType === 'مستأذن') attendanceStats.excuses++; } const newRecord = { title: actionName, score: scoreChange, timestamp: formattedTime }; student.actions = [newRecord, ...actionsList]; renderClassStudents(); await db.collection("allowed_students").doc(currentStudentId).update({ actions: student.actions, attendanceStats: attendanceStats }); } };
